    <!-- JavaScript logic -->
    <script>
        let cart = [];

        function addToCart(title, price) {
            cart.push({ title, price });
            updateCartUI();
        }

        function updateCartUI() {
            document.getElementById('cart-count').innerText = cart.length;
            let cartItemsDiv = document.getElementById('cart-items');
            cartItemsDiv.innerHTML = '';
            
            let total = 0;
            cart.forEach((item, index) => {
                total += item.price;
                cartItemsDiv.innerHTML += `<p>${index + 1}. ${item.title} - Rs. ${item.price}</p>`;
            });
            
            document.getElementById('cart-total').innerText = total;
        }

        function openCart() {
            document.getElementById('cart-modal').style.display = 'block';
        }

        function closeCart() {
            document.getElementById('cart-modal').style.display = 'none';
        }

        function sendOrderWhatsApp() {
            if (cart.length === 0) {
                alert("Your cart is empty!");
                return;
            }

            let paymentMethod = document.getElementById('payment-method').value;
            let message = "Hi RJ Collection! I want to place an order:\n\n";
            
            let total = 0;
            cart.forEach((item, index) => {
                message += `${index + 1}. ${item.title} (Rs. ${item.price})\n`;
                total += item.price;
            });

            message += `\n*Total Amount:* Rs. ${total}`;
            message += `\n*Payment Method:* ${paymentMethod}`;

            let encodedMessage = encodeURIComponent(message);
            window.open(`https://wa.me/923285113836?text=${encodedMessage}`, '_blank');
        }
    </script>
